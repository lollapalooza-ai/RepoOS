# --- DATA BRIDGE ---
def prep_inputs(*args, **kwargs):
    import torch
    import requests
    import json

    # 1. Fetch massive payload from the legacy shop API
    # In a real-world scenario, the API endpoint might be passed via kwargs
    response = requests.get("http://127.0.0.1:8080/api/v1/orders")
    orders = response.json()
    
    # 2. Extract the target values for summation
    values_list = [order["cart"]["total_value"] for order in orders]
    num_nodes = len(values_list)

    # 3. Create the 'node_vals' tensor (shape [N, 1])
    # This contains the actual values to be summed.
    node_vals = torch.tensor(values_list, dtype=torch.float32).unsqueeze(1)

    # 4. Create the sparse 'adj_csr' tensor (shape [1, N]) to perform summation.
    # This is a sparse representation of a single row vector of all ones: [1, 1, ..., 1].
    # Multiplying this by the column vector 'node_vals' performs a sum.
    crow_indices = torch.tensor([0, num_nodes], dtype=torch.int64)
    col_indices = torch.arange(num_nodes, dtype=torch.int64)
    values = torch.ones(num_nodes, dtype=torch.float32)
    adj_csr = torch.sparse_csr_tensor(crow_indices, col_indices, values, size=(1, num_nodes))

    # 5. Create the pre-allocated dense output tensor as required by the kernel.
    out = torch.zeros((1, 1), dtype=torch.float32)

    # 6. Return a list of tensors matching the 'forward' signature.
    return [adj_csr, node_vals, out]

def post_process(tensor_output, *args, **kwargs):
    import torch

    # The kernel's output is a tensor of shape [1, 1] containing the final sum.
    # The .item() method extracts the scalar value from the tensor.
    final_sum = tensor_output.item()

    return final_sum
